# Resource types allowed in infra/ — each one costs nothing on its own (ADR 0001).
# One type per line, then "#" and why it is free. A type missing here fails CI
# (scripts/check-free-tier.sh). Adding a line = a cost decision to review in the PR.

aws_cloudfront_distribution            # always free: 1 TB out + 10M requests / month
aws_cloudfront_origin_access_control   # configuration only, no charge
aws_cloudwatch_log_group               # always free: 5 GB ingestion + storage / month
aws_dynamodb_table                     # always free: 25 GB + 25 RCU / 25 WCU, PROVISIONED mode only (docs/data-model.md)
aws_iam_openid_connect_provider        # IAM is free
aws_iam_role                           # IAM is free
aws_iam_role_policy                    # IAM is free
aws_iam_role_policy_attachment         # IAM is free
aws_lambda_function                    # always free: 1M requests + 400k GB-s / month
aws_lambda_function_url                # no charge beyond Lambda invocations
aws_lambda_permission                  # configuration only, no charge
aws_resourcegroups_group               # Resource Groups are free
aws_scheduler_schedule                 # EventBridge Scheduler: 14M invocations / month free; hourly = ~720
aws_cognito_user_pool                  # Essentials/Lite: 10,000 MAU / month free, does not expire; no Plus add-ons, no SMS
aws_cognito_user_pool_client           # configuration only, no charge
aws_cognito_user_pool_domain           # Cognito prefix domain, no charge (no custom domain / certificate)
aws_sns_topic                          # always free: 1,000 e-mail deliveries / month (SNS FAQ); no KMS key
aws_sns_topic_subscription             # configuration only, no charge
aws_cloudwatch_metric_alarm            # always free: 10 standard-resolution alarms / month
